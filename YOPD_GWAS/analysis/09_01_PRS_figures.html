Create boxplots¶

Boxplots were created for different sets of summary statistics. Here an example.

# 1. Define groups by age at onset
if (!"Group_num" %in% colnames(prs_data)) {
  prs_data <- prs_data %>%
    mutate(
      Group_raw = as.character(Group),
      Group_num = case_when(
        Group_raw == "18_39" ~ 1L,
        Group_raw == "40_59" ~ 2L,
        Group_raw == "60_80" ~ 3L,
        TRUE ~ NA_integer_
      )
    )
}

# 2. Prepare data for vertical violin + boxplots (cases only; groups 1-3)
box_df <- prs_data %>%
  filter(PHENO == 2, Group_num %in% c(1,2,3), !is.na(PRS_std)) %>%
  mutate(
    Pathway = factor(Pathway, levels = pathway_order),
    Group = recode(as.character(Group_num), "1" = "18-39", "2" = "40-59", "3" = "60-80"),
    Group = factor(Group, levels = c("18-39","40-59","60-80"))
  )

# 3. Create boxplots
if (nrow(box_df) == 0) {
  message("No case-data available for plots")
} else {
  # y-limits
  y_min <- min(box_df$PRS_std, na.rm = TRUE)
  y_max <- max(box_df$PRS_std, na.rm = TRUE)
  y_range <- y_max - y_min
  if (!is.finite(y_range) || y_range == 0) y_range <- 1

  p_violin_box_vert <- ggplot(
    box_df,
    aes(x = Pathway, y = PRS_std, fill = Group, group = interaction(Pathway, Group))
  ) +
    geom_boxplot(
      position = position_dodge(width = 0.8),
      width = 0.5,
      outlier.size = 0.6,
      alpha = 0.85
    ) +
    # Mean as dashed line (drawn on top so it's visible in both violin and boxplot)
  stat_summary(
  fun = mean,
  geom = "errorbar",
  aes(ymin = after_stat(y), ymax = after_stat(y)),
  width = 0.7,
  position = position_dodge(width = 0.8),
  linetype = "dashed",
  color = "white",
  linewidth = 0.8
) +
    scale_fill_manual(values = c("60-80" = "#4DAF4A", "40-59" = "#377EB8", "18-39" = "#E41A1C")) +
    coord_cartesian(ylim = c(-5, 7)) +
    labs(
      x = "Pathway",
      y = "Standardized PRS (z-score vs controls)",
      fill = "Age group",
      title = "PRS per pathway (IPDGC) — cases only (violin + boxplots)"
    ) +
    theme_minimal(base_size = 13) +
    theme(
      axis.text.x = element_text(angle = 45, hjust = 1),
      legend.position = "right",
      plot.title = element_text(face = "bold")
    )

  print(p_violin_box_vert)
  ggsave(
    "...",
    plot = p_violin_box_vert,
    width = 12, height = 12
  )
}
